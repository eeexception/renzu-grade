cmake_minimum_required(VERSION 3.24)
if(CMAKE_HOST_APPLE AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
 set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum supported macOS version")
endif()
project(RenzuGrade VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(RENZU_ENABLE_METAL "Compile Apple Metal backend and shaders" ON)
option(RENZU_BUILD_TESTS "Build host-independent tests" ON)
option(RENZU_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
set(OFX_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/ofx" CACHE PATH "Official OpenFX SDK root (contains include/ofxImageEffect.h)")
if(NOT EXISTS "${OFX_SDK_ROOT}/include/ofxImageEffect.h")
 message(FATAL_ERROR "OFX_SDK_ROOT must contain include/ofxImageEffect.h")
endif()
if(RENZU_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
add_library(renzu_core STATIC src/color/Engine.cpp src/color/Photographic.cpp src/auto/Analysis.cpp src/cpu/Renderer.cpp
 src/match/Match.cpp src/match/ReferenceStore.cpp)
target_include_directories(renzu_core PUBLIC include)
target_compile_options(renzu_core PRIVATE -Wall -Wextra -Wpedantic)
find_package(Threads REQUIRED)
target_link_libraries(renzu_core PUBLIC Threads::Threads)
set(BUNDLE "${CMAKE_BINARY_DIR}/RenzuGrade.ofx.bundle")
file(MAKE_DIRECTORY "${BUNDLE}/Contents/MacOS" "${BUNDLE}/Contents/Resources")
configure_file(cmake/Info.plist.in "${BUNDLE}/Contents/Info.plist" @ONLY)
add_library(RenzuGrade MODULE src/ofx/Plugin.cpp)
target_include_directories(RenzuGrade PRIVATE "${OFX_SDK_ROOT}/include")
target_link_libraries(RenzuGrade PRIVATE renzu_core ${CMAKE_DL_LIBS})
target_compile_options(RenzuGrade PRIVATE -Wall -Wextra -Wpedantic -fvisibility=hidden)
set_target_properties(RenzuGrade PROPERTIES PREFIX "" SUFFIX ".ofx" LIBRARY_OUTPUT_DIRECTORY "${BUNDLE}/Contents/MacOS")
set_property(TARGET RenzuGrade APPEND PROPERTY LINK_DEPENDS "${BUNDLE}/Contents/Info.plist")
if(RENZU_ENABLE_METAL)
 if(NOT APPLE)
  message(FATAL_ERROR "Metal requires macOS. Pass -DRENZU_ENABLE_METAL=OFF explicitly for portable CPU tests.")
 endif()
 enable_language(OBJCXX)
 set(CMAKE_OBJCXX_STANDARD 17)
 execute_process(COMMAND xcrun -sdk macosx metal --version RESULT_VARIABLE METAL_STATUS OUTPUT_QUIET ERROR_QUIET)
 if(NOT METAL_STATUS EQUAL 0)
  message(FATAL_ERROR "Metal compiler unavailable. Install Xcode and run: xcodebuild -downloadComponent MetalToolchain. CPU-only builds require explicit -DRENZU_ENABLE_METAL=OFF.")
 endif()
 set(METALLIB "${BUNDLE}/Contents/Resources/renzu_grade.metallib")
 add_custom_command(OUTPUT "${METALLIB}"
  COMMAND xcrun -sdk macosx metal -std=metal3.0 "-mmacosx-version-min=${CMAKE_OSX_DEPLOYMENT_TARGET}" -fno-fast-math -I "${CMAKE_CURRENT_SOURCE_DIR}/include" -c "${CMAKE_CURRENT_SOURCE_DIR}/shaders/renzu_grade.metal" -o "${CMAKE_BINARY_DIR}/renzu_grade.air"
  COMMAND xcrun -sdk macosx metallib "${CMAKE_BINARY_DIR}/renzu_grade.air" -o "${METALLIB}"
  DEPENDS shaders/renzu_grade.metal include/renzu/Kernel.h VERBATIM)
 add_custom_target(renzu_shaders ALL DEPENDS "${METALLIB}")
 add_library(renzu_metal STATIC src/metal/MetalBackend.mm)
 target_link_libraries(renzu_metal PUBLIC renzu_core "-framework Metal" "-framework Foundation")
 target_compile_options(renzu_metal PRIVATE -fobjc-arc -Wall -Wextra)
 # renzu_metal already brings in renzu_core transitively.
 set_property(TARGET RenzuGrade PROPERTY LINK_LIBRARIES "renzu_metal;${CMAKE_DL_LIBS}")
 target_compile_definitions(RenzuGrade PRIVATE RENZU_METAL=1)
 add_dependencies(RenzuGrade renzu_shaders)
 # A changed resource invalidates the bundle signature even if C++ did not change.
 set_property(TARGET RenzuGrade APPEND PROPERTY LINK_DEPENDS "${METALLIB}")
endif()
if(APPLE)
 add_custom_command(TARGET RenzuGrade POST_BUILD COMMAND codesign --force --sign - "${BUNDLE}" VERBATIM)
endif()
if(RENZU_BUILD_TESTS)
 enable_testing()
 add_executable(renzu_tests tests/CoreTests.cpp)
 target_link_libraries(renzu_tests PRIVATE renzu_core)
 add_test(NAME color_and_analysis COMMAND renzu_tests)
 add_executable(renzu_match_tests tests/MatchTests.cpp)
 target_link_libraries(renzu_match_tests PRIVATE renzu_core)
 add_test(NAME reference_matching COMMAND renzu_match_tests)
 add_executable(renzu_photographic_tests tests/PhotographicTests.cpp)
 target_link_libraries(renzu_photographic_tests PRIVATE renzu_core)
 add_test(NAME photographic_controls COMMAND renzu_photographic_tests)
 add_executable(renzu_ofx_tests tests/OFXTests.cpp)
 target_include_directories(renzu_ofx_tests PRIVATE "${OFX_SDK_ROOT}/include")
 target_link_libraries(renzu_ofx_tests PRIVATE renzu_core ${CMAKE_DL_LIBS})
 add_dependencies(renzu_ofx_tests RenzuGrade)
 add_test(NAME ofx_host_contract COMMAND renzu_ofx_tests "$<TARGET_FILE:RenzuGrade>")
 if(RENZU_ENABLE_METAL)
  add_executable(renzu_metal_tests tests/MetalTests.mm)
  target_compile_options(renzu_metal_tests PRIVATE -fobjc-arc)
  target_link_libraries(renzu_metal_tests PRIVATE renzu_metal)
  add_dependencies(renzu_metal_tests renzu_shaders)
  add_test(NAME metal_cpu_parity COMMAND renzu_metal_tests "${METALLIB}")
  set_tests_properties(metal_cpu_parity PROPERTIES SKIP_RETURN_CODE 77)
 endif()
endif()
